cmake_minimum_required(VERSION 3.20)
project(coi_port C CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
# Link the C/C++ runtime statically on Windows, so coi.exe runs without the Visual C++
# Redistributable installed.
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")

set(THIRD_PARTY ${CMAKE_CURRENT_SOURCE_DIR}/../third_party)
if(NOT EXISTS ${THIRD_PARTY}/SDL/CMakeLists.txt)
    message(FATAL_ERROR "third_party/ is missing: run python tools/fetch_deps.py first")
endif()

# --- SDL3: a static library, or on Android the shared libSDL3.so -------------
# On Android the app is SDL's Java activity (android/), which loads libSDL3.so and then the
# port as libmain.so.
if(ANDROID)
    set(SDL_SHARED ON CACHE BOOL "" FORCE)
    set(SDL_STATIC OFF CACHE BOOL "" FORCE)
    set(SDL_LIBRARY SDL3::SDL3-shared)
else()
    set(SDL_SHARED OFF CACHE BOOL "" FORCE)
    set(SDL_STATIC ON CACHE BOOL "" FORCE)
    set(SDL_LIBRARY SDL3::SDL3-static)
endif()
set(SDL_TEST_LIBRARY OFF CACHE BOOL "" FORCE)
set(SDL_EXAMPLES OFF CACHE BOOL "" FORCE)
add_subdirectory(${THIRD_PARTY}/SDL ${CMAKE_BINARY_DIR}/SDL EXCLUDE_FROM_ALL)

# --- Musashi 68000 core ------------------------------------------------------
# m68kmake generates the opcode handlers from m68k_in.c at build time. Cross-compiling (the
# Android build) it can't run what it builds, so it uses the m68kmake of the desktop build
# (build/port), which has to be built first.
set(MUSASHI ${THIRD_PARTY}/Musashi)
set(MUSASHI_GEN ${CMAKE_BINARY_DIR}/musashi)
if(CMAKE_CROSSCOMPILING)
    find_program(M68KMAKE m68kmake NO_DEFAULT_PATH PATHS
        ${CMAKE_CURRENT_SOURCE_DIR}/../build/port/Release ${CMAKE_CURRENT_SOURCE_DIR}/../build/port/Debug)
    if(NOT M68KMAKE)
        message(FATAL_ERROR "Build the desktop port first (build/port): the Android build runs its m68kmake.")
    endif()
else()
    add_executable(m68kmake ${MUSASHI}/m68kmake.c)
    target_compile_definitions(m68kmake PRIVATE _CRT_SECURE_NO_WARNINGS)
    set(M68KMAKE m68kmake)
endif()
add_custom_command(
    OUTPUT ${MUSASHI_GEN}/m68kops.c ${MUSASHI_GEN}/m68kops.h
    COMMAND ${CMAKE_COMMAND} -E make_directory ${MUSASHI_GEN}
    COMMAND ${M68KMAKE} ${MUSASHI_GEN} ${MUSASHI}/m68k_in.c
    DEPENDS ${M68KMAKE} ${MUSASHI}/m68k_in.c
    COMMENT "Generating Musashi opcode handlers")

add_library(musashi STATIC
    ${MUSASHI}/m68kcpu.c
    ${MUSASHI}/m68kdasm.c
    ${MUSASHI}/softfloat/softfloat.c
    ${MUSASHI_GEN}/m68kops.c)
# Our configuration (src/musashi_conf.h) wraps Musashi's own m68kconf.h.
target_include_directories(musashi PUBLIC ${MUSASHI} ${MUSASHI_GEN} ${CMAKE_CURRENT_SOURCE_DIR}/src)
target_compile_definitions(musashi PUBLIC MUSASHI_CNF="musashi_conf.h" PRIVATE _CRT_SECURE_NO_WARNINGS)

# --- ymfm (YM2612) and the Z80 core ------------------------------------------
file(GLOB YMFM_SOURCES ${THIRD_PARTY}/ymfm/src/*.cpp)
add_library(ymfm STATIC ${YMFM_SOURCES})
target_include_directories(ymfm PUBLIC ${THIRD_PARTY}/ymfm/src)

add_library(z80 STATIC ${THIRD_PARTY}/z80/z80.c)
target_include_directories(z80 PUBLIC ${THIRD_PARTY}/z80)

# --- Dear ImGui (the F1 menu), with its SDL3 and OpenGL 3 backends ----------
set(IMGUI ${THIRD_PARTY}/imgui)
add_library(imgui STATIC
    ${IMGUI}/imgui.cpp
    ${IMGUI}/imgui_demo.cpp
    ${IMGUI}/imgui_draw.cpp
    ${IMGUI}/imgui_tables.cpp
    ${IMGUI}/imgui_widgets.cpp
    ${IMGUI}/backends/imgui_impl_sdl3.cpp
    ${IMGUI}/backends/imgui_impl_opengl3.cpp)
target_include_directories(imgui PUBLIC ${IMGUI} ${IMGUI}/backends)
target_compile_definitions(imgui PUBLIC IMGUI_DISABLE_DEMO_WINDOWS)
target_link_libraries(imgui PUBLIC ${SDL_LIBRARY})
if(ANDROID)
    target_link_libraries(imgui PUBLIC GLESv3)    # its OpenGL backend calls ES 3 directly there
endif()

# --- hqx (libretro glsl-shaders, LGPL 2.1 or later) -------------------------
# The two shader passes and the lookup tables are built into the program as a generated
# header; display.cpp compiles the shaders as they are.
set(HQX ${THIRD_PARTY}/glsl-shaders/hqx)
set(HQX_HEADER ${CMAKE_CURRENT_BINARY_DIR}/generated/hqx_data.h)
set(HQX_FILES ${HQX}/shader-files/hqx-pass1.glsl ${HQX}/shader-files/hqx-pass2.glsl
    ${HQX}/resources/hq2x.png ${HQX}/resources/hq3x.png ${HQX}/resources/hq4x.png)
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS ${HQX_FILES})
file(READ ${HQX}/shader-files/hqx-pass1.glsl HQX_PASS1)
file(READ ${HQX}/shader-files/hqx-pass2.glsl HQX_PASS2)
set(HQX_TEXT "// Generated by port/CMakeLists.txt from third_party/glsl-shaders/hqx.\n#pragma once\n\n")
string(APPEND HQX_TEXT "static const char kHqxPass1[] = R\"hqx(${HQX_PASS1})hqx\";\n")
string(APPEND HQX_TEXT "static const char kHqxPass2[] = R\"hqx(${HQX_PASS2})hqx\";\n")
foreach(N 2 3 4)
    file(READ ${HQX}/resources/hq${N}x.png PNG HEX)
    string(REGEX REPLACE "([0-9a-f][0-9a-f])" "0x\\1," PNG "${PNG}")
    string(APPEND HQX_TEXT "static const unsigned char kHq${N}xLut[] = {${PNG}};\n")
endforeach()
file(WRITE ${HQX_HEADER}.new "${HQX_TEXT}")
configure_file(${HQX_HEADER}.new ${HQX_HEADER} COPYONLY)    # rewritten only when it changes

# --- The app icon --------------------------------------------------------------
# The 32 x 32 master (icon/apple.png) built in for the window's icon; tools/make_icons.py makes
# the other platforms' icons from it (icon/coi.ico, built into coi.exe through icon/coi.rc).
set(ICON_HEADER ${CMAKE_CURRENT_BINARY_DIR}/generated/icon_data.h)
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS ${CMAKE_CURRENT_SOURCE_DIR}/icon/apple.png)
file(READ ${CMAKE_CURRENT_SOURCE_DIR}/icon/apple.png ICON_PNG HEX)
string(REGEX REPLACE "([0-9a-f][0-9a-f])" "0x\\1," ICON_PNG "${ICON_PNG}")
file(WRITE ${ICON_HEADER}.new "// Generated by port/CMakeLists.txt from port/icon/apple.png.\n#pragma once\n\n"
     "static const unsigned char kIconPng[] = {${ICON_PNG}};\n")
configure_file(${ICON_HEADER}.new ${ICON_HEADER} COPYONLY)

# --- The port ----------------------------------------------------------------
# coi.exe on the desktop; on Android, libmain.so for SDL's activity to load.
set(PORT_SOURCES
    src/main.cpp
    src/bus.cpp
    src/cheats.cpp
    src/interp.cpp
    src/mdplus.cpp
    src/display.cpp
    src/fade.cpp
    src/menu.cpp
    src/patches.cpp
    src/settings.cpp
    src/smooth_scroll.cpp
    src/smooth_swing.cpp
    src/boss_rush.cpp
    src/sound.cpp
    src/touch.cpp
    src/vdp.cpp)
if(ANDROID)
    add_library(coi SHARED ${PORT_SOURCES} src/android_mdplus.cpp src/android_pacing.cpp src/android_rom.cpp)
    set_target_properties(coi PROPERTIES OUTPUT_NAME main)
    # Swappy, the Android Games SDK's frame pacing (a Gradle dependency, through Prefab).
    find_package(games-frame-pacing REQUIRED CONFIG)
    target_link_libraries(coi PRIVATE log android EGL games-frame-pacing::swappy_static)
    # The performance log (frame rates, CPU and GPU time, every 2 s to adb logcat -s COI): off
    # in releases.
    option(COI_TIMING_LOG "Log Android performance figures" OFF)
    if(COI_TIMING_LOG)
        target_compile_definitions(coi PRIVATE COI_TIMING_LOG)
    endif()
else()
    add_executable(coi ${PORT_SOURCES} $<$<BOOL:${WIN32}>:icon/coi.rc>)
    set_target_properties(coi PROPERTIES WIN32_EXECUTABLE TRUE)    # no terminal window; --console shows one
    if(UNIX)
        # Linux: the C++ runtime built in, so the program needs only the system's C library
        # (built in Valve's Steam Runtime SDK, it needs glibc 2.29 or later: SteamOS, Bazzite, Ubuntu 20.04+).
        target_link_options(coi PRIVATE -static-libstdc++ -static-libgcc)
    endif()
endif()
target_link_libraries(coi PRIVATE musashi ymfm z80 imgui ${SDL_LIBRARY})
target_compile_definitions(coi PRIVATE _CRT_SECURE_NO_WARNINGS)
target_include_directories(coi PRIVATE ${CMAKE_CURRENT_BINARY_DIR}/generated)
